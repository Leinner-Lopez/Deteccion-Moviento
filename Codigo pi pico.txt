/*
 * ============================================================================
 *  CONTROLADOR DC TB6612FNG Canal B + RP2040-ZERO + WS2812B
 *  Versión 3.0 - Open Loop con Supervisión y Telemetría
 *  Motor DC 12V nominal, PWM efectivo máximo 9V (75% duty @ VM=12V)
 * ============================================================================
 */

#include <hardware/pwm.h>
#include <Adafruit_NeoPixel.h>
#include <math.h>

// ============ CONFIGURACIÓN DE PINES ============

#define BIN1_PIN        9       // GPIO -> Buffer -> TB6612FNG BIN1
#define BIN2_PIN        10      // GPIO -> Buffer -> TB6612FNG BIN2
#define PWMB_PIN        12      // PWM5B -> Buffer -> TB6612FNG PWMB
#define STBY_PIN        11      // GPIO -> Buffer -> TB6612FNG STBY
#define PIN_LEDS        18      // WS2812B (GP13: GPIO libre, sin conflictos)
#define SENSOR_HALL     17      // A3144, salida open-collector

// ============ PARÁMETROS DEL MOTOR ============

#define PWM_FREQ            25000
#define PWM_RESOLUTION      65535

#define VMOT_VOLTAGE        12.0f
#define MAX_VOLTAGE_EFFECT  12.0f
#define DEFAULT_VOLTAGE     9.0f
#define MIN_SPEED_RPM       0
#define MAX_SPEED_RPM       4000

// ============ PARÁMETROS DE CONTROL ============

#define CONTROL_FREQ_HZ     10000
#define UPDATE_INTERVAL_US  97

#define RAMP_DEFAULT_RPM_S  300.0f

// ============ SUPERVISOR Y TELEMETRÍA ============

#define HEARTBEAT_TIMEOUT_MS    2000
#define TELEMETRY_INTERVAL_MS   100

// ============ LEDs ============

#define NUM_LEDS_MAX        100     // Memoria reservada para la tira
#define NUM_LEDS_DEFAULT    97      // Cantidad fisica real (ajustable con LEDCOUNT)
#define LED_BRIGHTNESS_MAX  150

// ============ RENDER POV (persistencia de vision) ============
// La tira gira montada en el eje. La posicion angular se deduce del sensor
// Hall (1 pulso por vuelta), NO de las RPM pedidas: el motor no alcanza el
// setpoint por el peso de la tira, asi que solo sirve el tiempo medido.

// Ancho de banda WS2812B: 24 bits/LED @ 800 kHz. Con POV_LEDS=16 cada
// strip.show() tarda ~0,78 ms -> ~1280 refrescos/s. A 1500 RPM (25 vueltas/s)
// entran 1280/25 = 51 columnas; se usan 40 para dejar margen.
//
// Por que 16 LEDs y no los 97: el refresco es proporcional a los LEDs
// ENVIADOS. Con 97 costaria 3 ms (~330 Hz) y solo entrarian 12 columnas: en
// el borde dos columnas contiguas quedarian a 50 LEDs de distancia y
// cualquier figura se rompe. Con 16 LEDs la separacion en el borde es ~2
// LEDs, comparable al paso radial, y la silueta se lee.
#define POV_SECTORS         40      // Columnas angulares por vuelta (9 c/u)
#define POV_LEDS            16      // LEDs usados para el dibujo (radio)
#define ANIM_FRAMES         6       // Fotogramas por animacion
#define POV_STALL_FACTOR    2       // Sin pulso en 2 periodos -> no esta girando

#define PATTERN_RAINBOW     0
#define PATTERN_RINGS       1
#define PATTERN_SPOKES      2
#define PATTERN_SPIRAL      3
#define PATTERN_SOLID       255     // Color fijo, sin POV (prueba de cableado)

// ============ BUFFER SERIAL ============

#define CMD_BUF_SIZE        64

// ============ ENUMERACIONES ============

enum MotorState {
    STATE_IDLE,
    STATE_RAMP_UP,
    STATE_RUNNING,
    STATE_RAMP_DOWN,
    STATE_COAST,
    STATE_FAULT
};

// ============ VARIABLES GLOBALES MOTOR (Core 0) ============

MotorState motorState = STATE_IDLE;

bool motorDirection = true;         // true = CW, false = CCW
float voltageMax = DEFAULT_VOLTAGE;
float rampRateRPMs = RAMP_DEFAULT_RPM_S;

float rpmTarget = 0.0f;             // Setpoint final (positivo, magnitud)
float rpmRamped = 0.0f;             // Setpoint actual de la rampa (positivo, magnitud)
float rpmReal = 0.0f;               // RPM medidos por Hall (positivo, magnitud)
float desfase = 0.0f;               // rpmRamped - rpmReal

uint16_t dutyMax = 0;
int dutyCurrent = 0;

uint64_t lastUpdateTime = 0;
uint32_t lastTelemetryTime = 0;
uint32_t lastHeartbeatTime = 0;

char cmdBuffer[CMD_BUF_SIZE];
uint8_t cmdIndex = 0;

// ============ VARIABLES GLOBALES LEDS (Core 1) ============

Adafruit_NeoPixel strip(NUM_LEDS_MAX, PIN_LEDS, NEO_GRB + NEO_KHZ800);

volatile bool g_ledsEnabled = true;
volatile uint8_t g_ledSpeed = 128;

// Patron activo: lo selecciona el gesto desde la web (comando PATTERN <n>)
volatile uint8_t g_ledPattern = PATTERN_RAINBOW;

// Color del modo solido (PATTERN_SOLID)
volatile uint8_t g_ledR = 0;
volatile uint8_t g_ledG = 200;
volatile uint8_t g_ledB = 255;

// Cantidad de LEDs realmente montados en la tira
volatile uint8_t g_ledCount = NUM_LEDS_DEFAULT;
volatile bool g_ledCountDirty = false;

// Sector angular que se esta dibujando (para telemetria/diagnostico)
volatile uint16_t g_povSector = 0;
volatile bool g_povLocked = false;   // true = hay señal de Hall valida

// ============ VARIABLES ISR HALL ============

volatile uint32_t g_hallLastMicros = 0;
volatile uint32_t g_hallPeriodUs = 0;
volatile bool g_hallNewPulse = false;

// ============ FUNCIONES AUXILIARES ============

void motorStopCoast() {
    digitalWrite(BIN1_PIN, LOW);
    digitalWrite(BIN2_PIN, LOW);
    analogWrite(PWMB_PIN, 0);
    dutyCurrent = 0;
}

void motorSetDirectionAndPWM(int duty, bool cw) {
    if (cw) {
        digitalWrite(BIN1_PIN, HIGH);
        digitalWrite(BIN2_PIN, LOW);
    } else {
        digitalWrite(BIN1_PIN, LOW);
        digitalWrite(BIN2_PIN, HIGH);
    }
    analogWrite(PWMB_PIN, duty);
    dutyCurrent = duty;
}

float rampSetpoint(float current, float target, float maxDelta) {
    if (target > current + maxDelta) return current + maxDelta;
    if (target < current - maxDelta) return current - maxDelta;
    return target;
}

// ============ ISR SENSOR HALL ============

void onHallPulse() {
    uint32_t now = micros();
    uint32_t period = now - g_hallLastMicros;
    if (period > 1000) {
        g_hallPeriodUs = period;
        g_hallLastMicros = now;
        g_hallNewPulse = true;
    }
}

// ============ INICIALIZACIÓN PWM ============

void initPWM() {
    Serial.println("[PWM] Inicializando TB6612FNG Canal B...");

    pinMode(BIN1_PIN, OUTPUT);
    pinMode(BIN2_PIN, OUTPUT);
    pinMode(STBY_PIN, OUTPUT);
    pinMode(PWMB_PIN, OUTPUT);

    digitalWrite(STBY_PIN, HIGH);
    motorStopCoast();

    analogWriteFreq(PWM_FREQ);
    analogWriteResolution(16);

    uint slice_B = pwm_gpio_to_slice_num(PWMB_PIN);
    pwm_set_enabled(slice_B, false);
    pwm_set_counter(slice_B, 0);
    pwm_set_enabled(slice_B, true);

    Serial.println("[PWM] OK 25kHz en PWMB (GP11)");
}

void recalcDutyMax() {
    dutyMax = (uint16_t)((voltageMax / VMOT_VOLTAGE) * (float)PWM_RESOLUTION);
}

void setMotorVoltage(float v) {
    v = constrain(v, 0.0f, MAX_VOLTAGE_EFFECT);
    voltageMax = v;
    recalcDutyMax();
    Serial.print("OK VOLT ");
    Serial.println(voltageMax);
}

void setRampRate(float r) {
    r = constrain(r, 10.0f, 2000.0f);
    rampRateRPMs = r;
    Serial.print("OK RAMP ");
    Serial.println(rampRateRPMs);
}

void setMotorDirection(bool cw) {
    if (motorState != STATE_IDLE) {
        Serial.println("ERR DIR Solo modificable en IDLE");
        return;
    }
    motorDirection = cw;
    Serial.print("OK DIR ");
    Serial.println(cw ? "CW" : "CCW");
}

// ============ MÁQUINA DE ESTADOS DEL MOTOR ============

void enterFault(const char* reason) {
    motorState = STATE_FAULT;
    motorStopCoast();
    Serial.print("ERR FAULT ");
    Serial.println(reason);
}

void updateMotor() {
    // 1. Verificar heartbeat
    if (millis() - lastHeartbeatTime > HEARTBEAT_TIMEOUT_MS) {
        if (motorState != STATE_IDLE && motorState != STATE_FAULT) {
            enterFault("NO_HEARTBEAT");
            return;
        }
    }

    // 2. Calcular paso de rampa
    float rampStep = (rampRateRPMs * (float)UPDATE_INTERVAL_US) / 1000000.0f;

    switch (motorState) {
        case STATE_IDLE:
            return;

        case STATE_RAMP_UP:
            rpmRamped = rampSetpoint(rpmRamped, rpmTarget, rampStep);
            if (fabsf(rpmRamped - rpmTarget) < 1.0f) {
                rpmRamped = rpmTarget;
                motorState = STATE_RUNNING;
            }
            break;

        case STATE_RUNNING:
            if (rpmTarget > rpmRamped + 1.0f) {
                motorState = STATE_RAMP_UP;
            } else if (rpmTarget < rpmRamped - 1.0f) {
                motorState = STATE_RAMP_DOWN;
            }
            rpmRamped = rpmTarget;
            break;

        case STATE_RAMP_DOWN:
            rpmRamped = rampSetpoint(rpmRamped, 0.0f, rampStep);
            if (rpmRamped <= 1.0f) {
                rpmRamped = 0.0f;
                motorState = STATE_COAST;
            }
            break;

        case STATE_COAST:
            motorStopCoast();
            motorState = STATE_IDLE;
            Serial.println("OK COASTED");
            return;

        case STATE_FAULT:
            return;
    }

    // 3. Calcular RPM real desde Hall
    noInterrupts();
    uint32_t hallPeriod = g_hallPeriodUs;
    g_hallNewPulse = false;
    interrupts();

    if (hallPeriod > 0) {
        rpmReal = (60.0f * 1000000.0f) / (float)hallPeriod;
    } else {
        rpmReal = 0.0f;
    }

    desfase = rpmRamped - rpmReal;

    // 4. Calcular duty lineal: (rpmRamped / 4000) * dutyMax
    int duty = (int)((rpmRamped / (float)MAX_SPEED_RPM) * (float)dutyMax);
    if (duty > dutyMax) duty = dutyMax;
    if (duty < 0) duty = 0;

    // 5. Aplicar dirección y PWM
    if (rpmRamped < 0.5f) {
        motorStopCoast();
    } else {
        motorSetDirectionAndPWM(duty, motorDirection);
    }
}

// ============ TELEMETRÍA ============

void sendTelemetry() {
    float dutyPct = 0.0f;
    if (PWM_RESOLUTION > 0) {
        dutyPct = (dutyCurrent * 100.0f) / PWM_RESOLUTION;
    }

    Serial.print("TEL state=");
    switch (motorState) {
        case STATE_IDLE:      Serial.print("IDLE"); break;
        case STATE_RAMP_UP:   Serial.print("RAMP_UP"); break;
        case STATE_RUNNING:   Serial.print("RUNNING"); break;
        case STATE_RAMP_DOWN: Serial.print("RAMP_DOWN"); break;
        case STATE_COAST:     Serial.print("COAST"); break;
        case STATE_FAULT:     Serial.print("FAULT"); break;
    }
    Serial.print(" rpm_target="); Serial.print(rpmRamped, 1);
    Serial.print(" rpm_real=");  Serial.print(rpmReal, 1);
    Serial.print(" desfase=");   Serial.print(desfase, 1);
    Serial.print(" volt=");      Serial.print(voltageMax, 1);
    Serial.print(" duty=");      Serial.print(dutyPct, 1);
    Serial.print(" hall_dt=");    Serial.print(g_hallPeriodUs);
    Serial.print(" ramp_rate=");  Serial.print(rampRateRPMs, 1);
    // Campos nuevos al FINAL: los parsers viejos (GUI Python) los ignoran
    Serial.print(" pattern=");    Serial.print(g_ledPattern);
    Serial.print(" pov_sector="); Serial.print(g_povSector);
    Serial.print(" led_count=");  Serial.print(g_ledCount);
    Serial.println();
}

    Serial.print("STATUS state=");
    switch (motorState) {
        case STATE_IDLE: Serial.print("IDLE"); break;
        case STATE_RAMP_UP: Serial.print("RAMP_UP"); break;
        case STATE_RUNNING: Serial.print("RUNNING"); break;
        case STATE_RAMP_DOWN: Serial.print("RAMP_DOWN"); break;
        case STATE_COAST: Serial.print("COAST"); break;
        case STATE_FAULT: Serial.print("FAULT"); break;
    }
    Serial.print(" target="); Serial.print(rpmTarget);
    Serial.print(" ramped=");   Serial.print(rpmRamped);
    Serial.print(" real=");     Serial.print(rpmReal);
    Serial.print(" desfase="); Serial.print(desfase);
    Serial.print(" volt=");    Serial.print(voltageMax);
    Serial.print(" dir=");     Serial.print(motorDirection ? "CW" : "CCW");
    Serial.print(" ramp=");    Serial.print(rampRateRPMs);
    Serial.print(" leds=");    Serial.print(g_ledsEnabled ? "ON" : "OFF");

    Serial.print(",");          Serial.print(g_ledG);
    Serial.print(",");          Serial.println(g_ledB);
}

void processCommand(const char* cmd) {
    char buf[CMD_BUF_SIZE];
    strncpy(buf, cmd, CMD_BUF_SIZE - 1);
    buf[CMD_BUF_SIZE - 1] = '\0';

    char* token = strtok(buf, " ");
    if (token == NULL) return;

    for (char* p = token; *p; p++) *p = toupper(*p);

    char* arg = strtok(NULL, " ");

    if (strcmp(token, "START") == 0) {
        if (motorState == STATE_FAULT) {
            Serial.println("ERR En FAULT, enviar RESET primero");
            return;
        }
        int rpm = (arg != NULL) ? atoi(arg) : 100;
        rpm = constrain(rpm, MIN_SPEED_RPM, MAX_SPEED_RPM);
        rpmTarget = (float)rpm;

        if (motorState == STATE_IDLE) {
            initPWM();
            recalcDutyMax();
            motorState = STATE_RAMP_UP;
            rpmRamped = 0.0f;
            Serial.print("OK START ");
            Serial.println(rpm);
        } else {
            Serial.print("OK TARGET ");
            Serial.println(rpm);
        }
    }
    else if (strcmp(token, "STOP") == 0) {
        rpmTarget = 0.0f;
        if (motorState == STATE_RUNNING || motorState == STATE_RAMP_UP) {
            motorState = STATE_RAMP_DOWN;
            Serial.println("OK STOP");
        } else if (motorState == STATE_IDLE) {
            Serial.println("OK ALREADY_IDLE");
        }
    }
    else if (strcmp(token, "VOLT") == 0) {
        if (arg != NULL) {
            setMotorVoltage(atof(arg));
        } else {
            Serial.print("OK VOLT ");
            Serial.println(voltageMax);
        }
    }
    else if (strcmp(token, "RAMP") == 0) {
        if (arg != NULL) {
            setRampRate(atof(arg));
        } else {
            Serial.print("OK RAMP ");
            Serial.println(rampRateRPMs);
        }
    }
    else if (strcmp(token, "DIR") == 0) {
        if (arg != NULL) {
            setMotorDirection(strcmp(arg, "CW") == 0 || strcmp(arg, "cw") == 0);
        } else {
            Serial.print("OK DIR ");
            Serial.println(motorDirection ? "CW" : "CCW");
        }
    }
    else if (strcmp(token, "LEDS") == 0) {
        if (arg != NULL && (strcmp(arg, "ON") == 0 || strcmp(arg, "on") == 0)) {
            g_ledsEnabled = true;
            Serial.println("OK LEDS ON");
        } else if (arg != NULL && (strcmp(arg, "OFF") == 0 || strcmp(arg, "off") == 0)) {
            g_ledsEnabled = false;
            Serial.println("OK LEDS OFF");
        } else {
            Serial.print("OK LEDS ");
            Serial.println(g_ledsEnabled ? "ON" : "OFF");
        }
    }
    else if (strcmp(token, "LEDSPEED") == 0) {
        if (arg != NULL) {
            int sp = atoi(arg);
            sp = constrain(sp, 0, 255);
            g_ledSpeed = (uint8_t)sp;
            Serial.print("OK LEDSPEED ");
            Serial.println(g_ledSpeed);
        } else {
            Serial.print("OK LEDSPEED ");
            Serial.println(g_ledSpeed);
        }
    }
    else if (strcmp(token, "PATTERN") == 0) {
        // PATTERN <n> -> figura POV que selecciona el gesto desde la web
        // 0=RAINBOW, 1=WAVE, 2=RUNNER, 3=HEARTBEAT
        if (arg != NULL) {
            int p = constrain(atoi(arg), 0, 3);
            g_ledPattern = (uint8_t)p;
            g_ledsEnabled = true;
        }
        Serial.print("OK PATTERN ");
        Serial.println(g_ledPattern);
    }
    else if (strcmp(token, "LEDCOLOR") == 0) {
        // LEDCOLOR <r> <g> <b> -> color fijo sin POV, util para probar cableado
        char* argG = strtok(NULL, " ");
        char* argB = strtok(NULL, " ");
        if (arg != NULL && argG != NULL && argB != NULL) {
            g_ledR = (uint8_t)constrain(atoi(arg),  0, 255);
            g_ledG = (uint8_t)constrain(atoi(argG), 0, 255);
            g_ledB = (uint8_t)constrain(atoi(argB), 0, 255);
            g_ledsEnabled = true;
            g_ledPattern = PATTERN_SOLID;
        }
        Serial.print("OK LEDCOLOR ");
        Serial.print(g_ledR); Serial.print(" ");
        Serial.print(g_ledG); Serial.print(" ");
        Serial.println(g_ledB);
    }
    else if (strcmp(token, "LEDCOUNT") == 0) {
        // Calibracion: cantidad de LEDs realmente soldados en la tira
        if (arg != NULL) {
            int n = constrain(atoi(arg), 1, NUM_LEDS_MAX);
            g_ledCount = (uint8_t)n;
            g_ledCountDirty = true;
        }
        Serial.print("OK LEDCOUNT ");
        Serial.println(g_ledCount);
    }
    else if (strcmp(token, "HBT") == 0) {
        lastHeartbeatTime = millis();
        Serial.println("OK HBT");
    }
    else if (strcmp(token, "STATUS") == 0) {
        printStatus();
    }
    else if (strcmp(token, "RESET") == 0) {
        if (motorState == STATE_FAULT) {
            motorState = STATE_IDLE;
            rpmTarget = 0.0f;
            rpmRamped = 0.0f;
            Serial.println("OK RESET");
        } else {
            Serial.println("ERR No en FAULT");
        }
    }
    else if (strcmp(token, "HELP") == 0) {
        printHelp();
    }
    else {
        Serial.print("ERR UNKNOWN ");
        Serial.println(token);
    }
}

// ============ SETUP Y LOOP (CORE 0) ============

void setup() {
    Serial.begin(115200);
    delay(1000);

    pinMode(SENSOR_HALL, INPUT_PULLUP);
    attachInterrupt(digitalPinToInterrupt(SENSOR_HALL), onHallPulse, FALLING);

    initPWM();
    recalcDutyMax();

    lastHeartbeatTime = millis();

    Serial.println("\n========================================");
    Serial.println("  DC Motor TB6612FNG + WS2812B v3.0");
    Serial.println("========================================");
    Serial.print("Pines: BIN1=GP"); Serial.print(BIN1_PIN);
    Serial.print(" BIN2=GP"); Serial.print(BIN2_PIN);
    Serial.print(" PWMB=GP"); Serial.print(PWMB_PIN);
    Serial.print(" STBY=GP"); Serial.println(STBY_PIN);
    Serial.print("LEDs: "); Serial.print(POV_LEDS);
    Serial.print(" usados para POV (de "); Serial.print(g_ledCount);
    Serial.print(") @ GP"); Serial.println(PIN_LEDS);
    Serial.print("POV: "); Serial.print(POV_SECTORS);
    Serial.println(" columnas por vuelta");
    Serial.print("PWM: "); Serial.print(PWM_FREQ / 1000);
    Serial.println("kHz | Control 10kHz");
    Serial.print("Voltaje efectivo max: ");
    Serial.print(voltageMax);
    Serial.println("V (VM=12V)");
    Serial.print("Ramp: ");
    Serial.print(RAMP_DEFAULT_RPM_S);
    Serial.println(" RPM/s default");
    Serial.println("Heartbeat: 2s timeout | Telemetry: 10Hz");
    Serial.println("========================================");
    Serial.println("Enviar HELP para comandos");
    Serial.println("========================================\n");
}

void loop() {
    uint64_t now = micros();
    if ((now - lastUpdateTime) >= (uint64_t)UPDATE_INTERVAL_US) {
        lastUpdateTime += (uint64_t)UPDATE_INTERVAL_US;
        updateMotor();
    }

    uint32_t nowMs = millis();
    if (nowMs - lastTelemetryTime >= TELEMETRY_INTERVAL_MS) {
        lastTelemetryTime = nowMs;
        sendTelemetry();
    }

    while (Serial.available()) {
        char c = Serial.read();
        if (c == '\n' || c == '\r') {
            if (cmdIndex > 0) {
                cmdBuffer[cmdIndex] = '\0';
                processCommand(cmdBuffer);
                cmdIndex = 0;
            }
        } else if (cmdIndex < CMD_BUF_SIZE - 1) {
            cmdBuffer[cmdIndex++] = c;
        }
    }
}

// ============ SETUP1 Y LOOP1 (CORE 1 - LEDs) ============

// Contador global de fase para animaciones
uint32_t animationPhaseCounter = 0;

// ============ ANIMACIONES DINÁMICAS ============

// ============ ANIMACIONES POV PRE-RENDERIZADAS ============
// Cada tabla es [fotograma][columna angular][LED] con la intensidad 0-255 ya
// calculada. Se generan fuera del Pico dibujando la figura en un lienzo de
// 256x256 y promediando el area que cubre cada celda POV (antialiasing): por
// eso los bordes salen suaves en vez de dentados, y por eso no hace falta
// trigonometria ni bitmaps en tiempo de ejecucion, solo una lectura de tabla.

// Pinta la columna angular `sector` del fotograma `frame` tintada de (r,g,b).
void povDrawAnim(const uint8_t anim[][POV_SECTORS][POV_LEDS], uint8_t frame,
                 uint16_t sector, uint8_t r, uint8_t g, uint8_t b) {
    if (sector >= POV_SECTORS) sector = POV_SECTORS - 1;
    for (uint8_t i = 0; i < POV_LEDS; i++) {
        uint16_t v = pgm_read_byte(&anim[frame][sector][i]);
        strip.setPixelColor(i, strip.Color((r * v) / 255, (g * v) / 255, (b * v) / 255));
    }
}

// El fotograma avanza con el reloj, no con el sector: la figura debe quedar
// quieta durante una vuelta entera o se emborrona.
static inline uint8_t animFrame(uint16_t periodMs) {
    return (uint8_t)((millis() / periodMs) % ANIM_FRAMES);
}

}

    povDrawAnim(ANIM_RUNNER, animFrame(70), sector, 40, 255, 60);
}

// ANIMACIÓN 3: Corazon latiendo (PATTERN 3)
void patternHeartbeat(uint16_t sector, uint16_t n, uint32_t animationPhase) {
    povDrawAnim(ANIM_HEART, animFrame(110), sector, 255, 0, 25);
}

// ANIMACIÓN 4: Rainbow mega-dinámico con efecto pulsante (PATTERN 0)
void rainbowAdvanced(uint16_t sector, uint16_t n, uint32_t animationPhase) {
    n = POV_LEDS;
    static uint16_t drift = 0;
    drift += (uint16_t)(g_ledSpeed >> 1);  // Drift más rápido

    // Efecto de "respiración" del brillo
    float breathPhase = (float)animationPhase * 3.14159265f / 256.0f;
    uint8_t breathBrightness = 200 + (int8_t)(sin(breathPhase) * 55);

    uint16_t base = (uint16_t)(((uint32_t)sector * 65535UL) / POV_SECTORS) + drift;

    for (uint16_t i = 0; i < n; i++) {
        // Hue base con desplazamiento por LED
        uint16_t hue = base + (uint16_t)(((uint32_t)i * 16384UL) / n);  // Mayor desplazamiento

        // Obtener color HSV
        uint32_t color = strip.ColorHSV(hue, 255, breathBrightness);

        // Agregar variación radial (más brillante en el centro)
        uint8_t brightness = (breathBrightness * i) / (n * 2) + (breathBrightness / 2);
        brightness = constrain(brightness, 0, 255);

        // Mezclar con el color base
        strip.setPixelColor(i, strip.gamma32(color));
    }

    // Crear efecto de "onda" pulsante cada sector
    if ((animationPhase % 16) < 8) {
        for (uint16_t i = 0; i < n / 4; i++) {
            strip.setPixelColor(i, strip.Color(255, 255, 255));
        }
    }
}


void setup1() {
    strip.begin();
    // Acortar la tira es lo que hace viable el POV: strip.show() tarda en
    // proporcion a los LEDs enviados, no a los encendidos.
    strip.updateLength(POV_LEDS);
    strip.setBrightness(LED_BRIGHTNESS_MAX);
    strip.clear();
    strip.show();
}

// ---------- Posicion angular a partir de la velocidad MEDIDA ----------
//
// El Hall entrega un pulso por vuelta. g_hallLastMicros marca el instante del
// pulso indice y g_hallPeriodUs cuanto duro la ultima vuelta REAL. El angulo
// actual es simplemente la fraccion de periodo transcurrida desde el indice:
//
//     sector = (micros() - g_hallLastMicros) * POV_SECTORS / g_hallPeriodUs
//
// Nunca se usa rpmTarget ni rpmRamped: si el motor no llega al setpoint por el
// peso de la tira, la figura igual queda anclada al eje porque el periodo que
// se divide es el que efectivamente tardo la vuelta.
//
// Ambas variables son uint32_t: en Cortex-M0+ cada lectura es atomica, y la
// ISR corre en el Core 0 mientras esto corre en el Core 1, asi que refrescar
// la tira no interfiere con la medicion.

uint16_t povCurrentSector() {
    uint32_t lastUs = g_hallLastMicros;
    uint32_t period = g_hallPeriodUs;
    uint32_t dt     = micros() - lastUs;

    if (period == 0 || dt > period * POV_STALL_FACTOR) {
        // Motor detenido o Hall sin señal: barrido lento para no quedar a oscuras
        g_povLocked = false;
        return (uint16_t)((millis() / 40) % POV_SECTORS);
    }

    g_povLocked = true;
    if (dt >= period) dt = period - 1;
    return (uint16_t)(((uint64_t)dt * POV_SECTORS) / period);
}

void loop1() {
    if (g_ledCountDirty) {
        g_ledCountDirty = false;
        strip.clear();
        strip.show();
    }

    if (!g_ledsEnabled) {
        strip.clear();
        strip.show();
        delay(50);
        return;
    }

    uint8_t pattern = g_ledPattern;
    uint16_t n = POV_LEDS;   // el dibujo ocupa los 32 LEDs interiores

    // Incrementar fase de animación (0-255 cíclico)
    animationPhaseCounter = (animationPhaseCounter + 2) % 256;

    if (pattern == PATTERN_SOLID) {
        strip.fill(strip.Color(g_ledR, g_ledG, g_ledB), 0, POV_LEDS);
        strip.show();
        delay(20);
        return;
    }

    // Sincronizar con rotación real
    uint16_t sector = povCurrentSector();
    g_povSector = sector;

    // Renderizar patrón con animación
    switch (pattern) {
        case PATTERN_RINGS:  // 1 - Mano saludando
            patternWave(sector, n, animationPhaseCounter);
            break;

        case PATTERN_SPOKES:  // 2 - Persona corriendo
            patternRunner(sector, n, animationPhaseCounter);
            break;

        case PATTERN_SPIRAL:  // 3 - Latido cardiaco
            patternHeartbeat(sector, n, animationPhaseCounter);
            break;

        case PATTERN_RAINBOW:  // 0 - Arcoíris
        default:
            rainbowAdvanced(sector, n, animationPhaseCounter);
            break;
    }

    strip.show();
}
